    import { Controller, Get, Param, Post } from '@nestjs/common';

@Controller('first')
export class FirstController {
  // GET /first
  @Get()
  index(): string {
    return 'Action [index] of the [first] controller';
  }

  // GET /first/action01
  @Get('action01')
  action01(): string {
    return 'Action [action01] of the [first] controller, GET request';
  }

  // POST /first/action01: hetzelfde pad, een andere HTTP-methode
  @Post('action01')
  action01Post(): string {
    return 'Action [action01Post] of the [first] controller, POST request';
  }

  // GET /first/products/17: [:id] is een routeparameter
  @Get('products/:id')
  product(@Param('id') id: string): string {
    return `Product no. ${id}`;
  }

  // GET /first/products/17/reviews/3: twee routeparameters
  @Get('products/:id/reviews/:number')
  review(@Param() params: { id: string; number: string }): string {
    return `Review no. ${params.number} of product no. ${params.id}`;
  }

  // GET /first/files/...: [*] accepteert elk einde van het pad
  @Get('files/*')
  file(@Param('0') path: string): string {
    return `Requested file: ${path}`;
  }
}

    We lichten deze code toe:

    • regel 3: [@Controller('first')] legt het voorvoegsel vast dat alle routes van de controller gemeen hebben: de acties ervan handelen URL-routes af die beginnen met [/first];
    • regel 6–7: [@Get()] zonder argumenten: de actie [index] handelt [GET /first] af;
    • regel 12–13: de actie [action01] verwerkt [GET /first/action01]. De URL is de samenvoeging van het voorvoegsel van de controller en het pad van de actie;
    • regel 18–19: hetzelfde pad, maar voor de methode POST: [@Post] vervangt [@Get]. Er zijn ook [@Put], [@Delete], [@Patch], [@All]... De naam van de [TypeScript] -methode ([action01Post]) is willekeurig: voor de routing zijn alleen de decorators van belang;
    • Regel 24–25: [:id] is een routeparameter: dit segment van de URL kan elke waarde aannemen. De decorator [@Param('id')] injecteert de waarde ervan in de parameter [id] van de actie. Deze waarde is altijd een tekenreeks: bij [/first/products/17] is [id] gelijk aan ['17'], niet het getal 17;
    • regel 30–31: twee routeparameters. [@Param()] zonder argument injecteert een object dat ze allemaal bevat: [{ id: '17', number: '3' }];
    • regel 36–37: het teken [*] accepteert elk einde van een URL, ook [/]. Het overeenkomende deel is de parameter met de naam [0].

    import { Controller, Get, Header, HttpCode, NotFoundException } from '@nestjs/common';

@Controller('responses')
export class ResponsesController {
  // een string: ongewijzigd verzonden, met het type [text/html]
  @Get('html')
  html(): string {
    return '<h1>Action [html]</h1>';
  }

  // dezelfde string, maar aangekondigd als platte tekst
  @Get('text')
  @Header('Content-Type', 'text/plain; charset=utf-8')
  text(): string {
    return '<h1>Action [text]</h1>';
  }

  // een object: geserialiseerd naar JSON, met het type [application/json]
  @Get('json')
  json(): object {
    return { lastName: 'Martin', firstName: 'Jean', age: 42 };
  }

  // een array: ook geserialiseerd naar JSON
  @Get('array')
  array(): string[] {
    return ['Mme', 'Mlle', 'Mr'];
  }

  // een gekozen statuscode: 201 (Created) in plaats van 200 (OK)
  @Get('created')
  @HttpCode(201)
  created(): object {
    return { message: 'resource created' };
  }

  // geen antwoord: de code 204 (No Content) betekent "niets weer te geven"
  @Get('empty')
  @HttpCode(204)
  empty(): void {}

  // een asynchrone methode: NestJS wacht tot de promise is afgehandeld
  @Get('promise')
  async promise(): Promise<object> {
    const start = Date.now();
    await new Promise((done) => setTimeout(done, 500));
    return { wait: `${Date.now() - start} ms` };
  }

  // een exception: NestJS zet die om in een foutantwoord
  @Get('error')
  error(): string {
    throw new NotFoundException('This resource does not exist');
  }
}

    We lichten deze code toe:

    • regel 6–9: een tekenreeks wordt ongewijzigd verzonden, met de header [Content-Type: text/html; charset=utf-8];
    • regel 12–16: de decorator [@Header] stelt een header van het antwoord in. Als de tekenreeks als platte tekst wordt gedeclareerd ([text/plain]), wordt ze door de browser niet meer geïnterpreteerd;
    • regel 19–22: een object wordt als JSON geserialiseerd, met de header [Content-Type: application/json];
    • regel 25–28: ook een array wordt als JSON geserialiseerd;
    • regel 31–35: [@HttpCode(201)] wijzigt de statuscode van het antwoord (standaard 200 bij een GET);
    • regel 38–40: de code 204 (No Content) betekent “geen inhoud”: de actie geeft niets terug;
    • regel 43–48: een actie kan asynchroon ([async]) zijn: in dat geval geeft ze een promise ([Promise]) terug en wacht [NestJS] tot die is afgehandeld voordat het antwoordt. Dit geldt voor alle acties die een database raadplegen;
    • Regel 51–54: een actie kan een exception veroorzaken. De exceptions van [NestJS] ([NotFoundException], [BadRequestException], [ForbiddenException]...) bevatten elk een HTTP-statuscode (hier 404), die [NestJS] voor zijn antwoord gebruikt.

    import { Controller, Get, Param, Redirect } from '@nestjs/common';

@Controller()
export class RedirectsController {
  // vaste redirect: het adres en de code zijn vooraf bekend
  @Get('old')
  @Redirect('/new', 301)
  oldPage(): void {}

  @Get('new')
  newPage(): string {
    return 'You are on the [new] page';
  }

  // berekende redirect: het geretourneerde object vervangt het adres van @Redirect()
  @Get('go/:destination')
  @Redirect()
  go(@Param('destination') destination: string): { url: string } {
    return { url: `/page/${destination}` };
  }

  @Get('page/:name')
  page(@Param('name') name: string): string {
    return `You are on the [${name}] page`;
  }

  // redirect naar een andere site
  @Get('documentation')
  @Redirect('https://docs.nestjs.com')
  documentation(): void {}
}

    We lichten deze code toe:

    • regel 6–8: [@Redirect('/new', 301)]: een permanente omleiding. De code 301 (Moved Permanently) geeft aan dat het adres van de pagina definitief is gewijzigd; de standaardcode is 302 (Found), een tijdelijke omleiding;
    • regel 16–20: een berekende omleiding: [@Redirect()] zonder argument, waarbij de actie een object [{ url: ... }] teruggeeft dat het doeladres bepaalt;
    • regel 28–30: u kunt ook omleiden naar een andere site.

    import { Controller, Get, Req, Res } from '@nestjs/common';
import { Request, Response } from 'express';

@Controller('express')
export class ExpressController {
  // het Express-verzoek: methode, URL, headers...
  @Get('request')
  request(@Req() req: Request): object {
    return {
      method: req.method,
      url: req.url,
      path: req.path,
      browser: req.get('user-agent'),
    };
  }

  // het Express-antwoord: de actie verstuurt het antwoord zelf
  @Get('response')
  response(@Res() res: Response): void {
    res.status(200).type('text/plain').send('Response sent by [res.send]');
  }

  // redirect met het Express-antwoord
  @Get('redirect')
  redirect(@Res() res: Response): void {
    res.redirect('/express/response');
  }

  // passthrough: we gebruiken het antwoord (hier voor een header), maar NestJS
  // verstuurt de door de actie geretourneerde waarde nog steeds zelf
  @Get('passthrough')
  passthrough(@Res({ passthrough: true }) res: Response): string {
    res.setHeader('X-Example', '05-express');
    return 'Response sent by NestJS, with an added header';
  }

  // FOUT om te vermijden: @Res() zonder passthrough, en er wordt niets verstuurd
  @Get('blocked')
  blocked(@Res() res: Response): string {
    return 'This string will never be sent: the browser keeps waiting...';
  }
}

    We lichten deze code toe:

    • regel 2: de typen [Request] en [Response] van [Express];
    • Regel 7–15: [@Req()] injecteert het verzoek van [Express]: HTTP-methode, URL, pad, headers ([req.get(...)])... ;
    • regel 18–21: [@Res()] injecteert het antwoord van [Express]. Let op: zodra een actie [@Res()] gebruikt, [NestJS] gaat het ervan uit dat die actie het antwoord verstuurt, en negeert het wat ze teruggeeft. Hier stelt de actie de statuscode en het type in en verstuurt ze de body met [res.send];
    • regel 24–27: een 302-omleiding die door [Express] wordt uitgevoerd met [res.redirect(url)];
    • regel 31–35: met de optie [passthrough: true] gebruikt de actie het antwoord (hier om een header toe te voegen), maar laat ze [NestJS] verzenden wat ze teruggeeft. Zo combineert u het beste van twee werelden;
    • regel 38–41: de fout die u moet vermijden: [@Res()] zonder [passthrough] en zonder aanroepen van [res.send], [res.redirect] of [res.render].... De teruggegeven tekenreeks wordt genegeerd, er wordt nooit een antwoord verzonden en de browser wacht eindeloos.

    import { Injectable } from '@nestjs/common';

export interface Product {
  id: number;
  name: string;
  price: number;
}

@Injectable()
export class ProductsService {
  private readonly products: Product[] = [
    { id: 1, name: 'pen', price: 1.5 },
    { id: 2, name: 'notebook', price: 3.2 },
    { id: 3, name: 'eraser', price: 0.8 },
  ];

  findAll(): Product[] {
    return this.products;
  }

  findById(id: number): Product | undefined {
    return this.products.find((p) => p.id === id);
  }
}

    • regel 11–15: de gegevens, hier in het geheugen;
    • regel 17–23: de twee methoden van de service.

    import { Controller, Get, NotFoundException, Param } from '@nestjs/common';
import { Product, ProductsService } from './products.service';

@Controller('products')
export class ProductsController {
  // dependency injection: NestJS levert de instantie van ProductsService
  constructor(private readonly productsService: ProductsService) {}

  @Get()
  list(): Product[] {
    return this.productsService.findAll();
  }

  @Get(':id')
  oneProduct(@Param('id') id: string): Product {
    const product = this.productsService.findById(Number(id));
    if (!product) {
      throw new NotFoundException(`Product no. ${id} not found`);
    }
    return product;
  }
}
    • regel 7: dependency injection. De controller geeft in zijn constructor aan dat hij een [ProductsService] nodig heeft; [NestJS] levert hem de enige instantie van de service. De controller roept nooit [new ProductsService()] aan. De syntaxis [private readonly productsService: ProductsService] van [TypeScript] declareert zowel de parameter als het veld [this.productsService];
    • Regel 9–12: de actie delegeert naar de service;
    • regel 14–21: [Number(id)] zet de routeparameter (een tekenreeks) om in een getal. Een niet-bestaand product veroorzaakt een 404-exception.

    import { Controller, Get } from '@nestjs/common';
import { ProductsService } from '../products/products.service';

@Controller('clients')
export class ClientsController {
  // een service uit een ANDERE module, geëxporteerd door die module
  constructor(private readonly productsService: ProductsService) {}

  @Get('cart')
  cart(): object {
    const products = this.productsService.findAll();
    const total = products.reduce((sum, p) => sum + p.price, 0);
    return { client: 'Jean Martin', items: products.length, total };
  }
}
